# Advanced Token Bucket Implementation class TokenBucket: def __init__(self, rate: float, capacity: int): self.rate = rate self.capacity = capacity self.tokens = capacity self.last_update = time.monotonic() self._lock = threading.Lock() def consume(self, tokens: int = 1) -> Tuple[bool, float]: with self._lock: now = time.monotonic() elapsed = now - self.last_update # Precise token generation self.tokens = min( self.capacity, self.tokens + elapsed * self.rate ) self.last_update = now if self.tokens >= tokens: self.tokens -= tokens return True, 0 # Calculate exact wait time deficit = tokens - self.tokens wait_time = deficit / self.rate return False, wait_time
